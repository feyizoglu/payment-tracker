## Shared API (`shared/bill-hunt.ts`)

Coordinates: y points down; world 70 wide, the floor at y = 90; cell c = row * 7 + col covers x ∈ [col·10, col·10+10), y ∈ [row·10, row·10+10).

```ts
export const HUNT_COLS = 7, HUNT_ROWS = 9, HUNT_CELL = 10;
export const HUNT_W = 70, HUNT_H = 90, HUNT_FOOT = 8;      // FOOT: room under the floor for the launcher
export const COIN_R = 1.2, COIN_SPEED = 90, LAUNCH_GAP = 0.08;
export const SUB_DT = 1 / 240, MAX_DT = 1 / 30;
export const MIN_ANGLE = (8 * Math.PI) / 180, MIN_PULL = 4;
export const SPEED_UP = 3, THROW_CAP = 30, COIN_CAP = 120;
export const POWER_CHANCE = 0.4, DOUBLE_CHANCE = 0.2, CAMPAIGN_HIT = 3;
export const BILL_GAP = 0.6;                                // a bill's box is its cell inset by this
export const HUNT_BEST_KEY = "bill-hunt-best", HUNT_OVER_GUARD_MS = 500;

export type HuntPhase = "ready" | "aiming" | "throwing" | "over";
export type HuntItemKind = "bill" | "plus" | "rowCut" | "colCut" | "deflect" | "campaign";
export interface HuntItem { id: number; kind: HuntItemKind; col: number; row: number; value: number; double: boolean; used: boolean }
export interface HuntCoin { x: number; y: number; vx: number; vy: number; down: boolean; cell: number }
export interface HuntThrow { coins: HuntCoin[]; total: number; fired: number; t: number; acc: number; speed: 1 | 3; firstDown: number | null; gained: number; angle: number }
export interface HuntState { phase: HuntPhase; month: number; score: number; coins: number; launcher: number; items: HuntItem[]; nextId: number; shot: HuntThrow | null; version: number }

newHuntGame(rng): HuntState            // ready, month 1, 1 coin, launcher 35, row spawned in row 0
huntStart(s): HuntState                 // ready -> aiming
huntAim(dx, dy): number | null          // pull vector (world units, y down) -> throw angle above horizontal
huntAimEnd(s, angle): { x, y }          // ghost coin to its first bounce / floor
huntFire(s, angle): HuntState           // aiming -> throwing
stepHunt(s, dt, rng): HuntState         // advances a throw; ends the month when every coin is down or THROW_CAP passed
huntSpeedUp(s): HuntState
huntCallBack(s, rng): HuntState         // ends the throw now
huntViewport(w, h): { scale, left, top }
huntBillShade(value, month, dark): string
HUNT_COLORS: { light, dark }
```

`version` goes up whenever an item changes or disappears, so screens re-render the board only then.
