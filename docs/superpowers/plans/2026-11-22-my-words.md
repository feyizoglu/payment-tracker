# My Words Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A per-user English vocabulary notebook with dictionary lookup and a three-in-a-row review, on web and mobile, as specified in `docs/superpowers/specs/2026-11-22-my-words-design.md`.

**Architecture:** Pure rules (validation, filters, review order, review session) live in `shared/vocab.ts` and `shared/vocab-review.ts` and are used by both apps. The server owns the dictionary (`src/lib/word-lookup.ts`, dictionaryapi.dev) and the `vocab_words` table through `/api/words`; the streak update is one Postgres function so two quick answers cannot race. Web pages under `src/app/words/`, mobile screens under `mobile/src/app/words/`, both thin over the shared modules.

**Tech Stack:** Next.js (App Router, route handlers with `params: Promise<…>`), Supabase (service role via `supabaseAdmin`), Expo Router 57 + NativeWind, vitest on both sides.

**Conventions:** branch `feature/vocab-words`; conventional commits; stage files by explicit path and run `git diff --cached --name-only` before each commit; every file ends with a newline. Web tests: `npx vitest run <file>` from the repo root. Mobile tests: `npx vitest run <file>` from `mobile/`. Mobile type check: `npx tsc --noEmit -p .` from `mobile/`. Web type check: `npx tsc --noEmit` from the root.

```ts
// shared/vocab.ts
import type { Translations } from "./i18n";

export type VocabLookup = "found" | "not_found" | "failed";

export interface VocabSense {
  pos: string;
  definition: string;
  example?: string;
}

export interface VocabWord {
  id: string;
  user_id: string;
  word: string;
  word_key: string;
  meaning_tr: string | null;
  ipa: string | null;
  audio_url: string | null;
  senses: VocabSense[];
  lookup: VocabLookup;
  streak: number;
  learned_at: string | null;
  reviewed_at: string | null;
  created_at: string;
}

export const VOCAB_WORD_MAX = 40;
export const VOCAB_MEANING_MAX = 200;
export const VOCAB_LEARNED_STREAK = 3;
export const VOCAB_ROUND_SIZE = 10;

const WORD_PATTERN = /^[A-Za-z]+(?:[ '-][A-Za-z]+)*$/;

export function normalizeWord(raw: string): string {
  return raw.replace(/[‘’]/g, "'").trim().replace(/\s+/g, " ");
}

export function isValidWord(word: string): boolean {
  return word.length <= VOCAB_WORD_MAX && WORD_PATTERN.test(word);
}

export function vocabWordKey(word: string): string {
  return word.toLowerCase();
}

export function normalizeMeaning(raw: unknown): { ok: true; value: string | null } | { ok: false } {
  if (raw === undefined || raw === null) return { ok: true, value: null };
  if (typeof raw !== "string") return { ok: false };
  const value = raw.trim();
  if (value.length > VOCAB_MEANING_MAX) return { ok: false };
  return { ok: true, value: value === "" ? null : value };
}

export function vocabProgress(words: readonly VocabWord[]): { total: number; learned: number } {
  return { total: words.length, learned: words.filter((w) => w.learned_at !== null).length };
}

export type VocabFilter = "all" | "learning" | "learned";

export function filterWords(words: readonly VocabWord[], filter: VocabFilter, query: string): VocabWord[] {
  const q = query.trim().toLocaleLowerCase("tr");
  return words.filter((w) => {
    if (filter === "learning" && w.learned_at !== null) return false;
    if (filter === "learned" && w.learned_at === null) return false;
    if (!q) return true;
    return w.word.toLocaleLowerCase("tr").includes(q) || (w.meaning_tr ?? "").toLocaleLowerCase("tr").includes(q);
  });
}

/** The words a review round asks: unlearned, lowest streak, never reviewed, longest unreviewed, oldest. */
export function reviewRound(words: readonly VocabWord[], size = VOCAB_ROUND_SIZE): VocabWord[] {
  return words
    .filter((w) => w.learned_at === null)
    .sort(
      (a, b) =>
        a.streak - b.streak ||
        (a.reviewed_at ?? "").localeCompare(b.reviewed_at ?? "") ||
        a.created_at.localeCompare(b.created_at)
    )
    .slice(0, size);
}

export function senseGroups(senses: readonly VocabSense[]): { pos: string; senses: VocabSense[] }[] {
  const groups: { pos: string; senses: VocabSense[] }[] = [];
  for (const s of senses) {
    const group = groups.find((g) => g.pos === s.pos);
    if (group) group.senses.push(s);
    else groups.push({ pos: s.pos, senses: [s] });
  }
  return groups;
}

export type VocabErrorCode = "invalid_word" | "invalid_meaning" | "duplicate" | "not_found" | "invalid_request";

export function vocabErrorText(code: unknown, fallback: string, t: Translations): string {
  switch (code) {
    case "invalid_word":
      return t.wordsInvalidWord;
    case "invalid_meaning":
      return t.wordsInvalidMeaning;
    case "duplicate":
      return t.wordsDuplicate;
    case "not_found":
      return t.wordsNotFound;
    default:
      return fallback || t.updateFailed;
  }
}
```

```ts
// shared/vocab-review.ts
import { VOCAB_LEARNED_STREAK, reviewRound, type VocabWord } from "./vocab";

export interface ReviewState {
  words: VocabWord[];
  index: number;
  revealed: boolean;
  known: number;
  learned: number;
}

export function startReview(all: readonly VocabWord[]): ReviewState {
  return { words: reviewRound(all), index: 0, revealed: false, known: 0, learned: 0 };
}

export const reviewDone = (s: ReviewState): boolean => s.index >= s.words.length;

export const currentWord = (s: ReviewState): VocabWord | null => s.words[s.index] ?? null;

export const revealReview = (s: ReviewState): ReviewState => (reviewDone(s) ? s : { ...s, revealed: true });

export function answerReview(s: ReviewState, know: boolean): ReviewState {
  const word = currentWord(s);
  if (!word) return s;
  return {
    ...s,
    index: s.index + 1,
    revealed: false,
    known: s.known + (know ? 1 : 0),
    learned: s.learned + (know && word.streak + 1 >= VOCAB_LEARNED_STREAK ? 1 : 0),
  };
}
```

```ts
import type { VocabLookup, VocabSense } from "@/lib/vocab";

// The only code that knows where definitions come from; a later Claude source replaces this file.
const DICTIONARY_URL = "https://api.dictionaryapi.dev/api/v2/entries/en/";
const TIMEOUT_MS = 5000;
const SENSES_PER_POS = 3;

export interface LookupResult {
  lookup: VocabLookup;
  ipa: string | null;
  audioUrl: string | null;
  senses: VocabSense[];
}

const empty = (lookup: VocabLookup): LookupResult => ({ lookup, ipa: null, audioUrl: null, senses: [] });

type Rec = Record<string, unknown>;
const isRec = (v: unknown): v is Rec => typeof v === "object" && v !== null;
const text = (v: unknown): string | null => (typeof v === "string" && v.trim() !== "" ? v.trim() : null);
const list = (v: unknown): Rec[] => (Array.isArray(v) ? v.filter(isRec) : []);

export function parseEntries(body: unknown): LookupResult | null {
  const entries = list(body);
  if (entries.length === 0) return null;
  const phonetics = entries.flatMap((e) => list(e.phonetics));
  const ipa = [...entries.map((e) => text(e.phonetic)), ...phonetics.map((p) => text(p.text))].find((s) => s !== null) ?? null;
  const audios = phonetics.map((p) => text(p.audio)).filter((a): a is string => a !== null && a.startsWith("https://"));
  const audioUrl = audios.find((a) => a.endsWith("-us.mp3")) ?? audios[0] ?? null;
  const senses: VocabSense[] = [];
  const perPos = new Map<string, number>();
  for (const meaning of entries.flatMap((e) => list(e.meanings))) {
    const pos = text(meaning.partOfSpeech) ?? "";
    for (const d of list(meaning.definitions)) {
      const definition = text(d.definition);
      if (!definition || (perPos.get(pos) ?? 0) >= SENSES_PER_POS) continue;
      perPos.set(pos, (perPos.get(pos) ?? 0) + 1);
      const example = text(d.example);
      senses.push(example ? { pos, definition, example } : { pos, definition });
    }
  }
  return { lookup: "found", ipa, audioUrl, senses };
}

export async function lookupWord(word: string, fetchImpl: typeof fetch = fetch): Promise<LookupResult> {
  let res: Response;
  try {
    res = await fetchImpl(DICTIONARY_URL + encodeURIComponent(word), { signal: AbortSignal.timeout(TIMEOUT_MS) });
  } catch {
    return empty("failed");
  }
  if (res.status === 404) return empty("not_found");
  if (!res.ok) return empty("failed");
  const body: unknown = await res.json().catch(() => null);
  return parseEntries(body) ?? empty("failed");
}
```

```sql
create table if not exists vocab_words (
  id uuid default gen_random_uuid() primary key,
  user_id uuid not null references users(id) on delete cascade,
  word text not null check (length(word) between 1 and 40),
  word_key text not null,
  meaning_tr text check (meaning_tr is null or length(meaning_tr) <= 200),
  ipa text,
  audio_url text,
  senses jsonb not null default '[]'::jsonb,
  lookup text not null default 'failed' check (lookup in ('found', 'not_found', 'failed')),
  streak int not null default 0 check (streak between 0 and 3),
  learned_at timestamptz,
  reviewed_at timestamptz,
  created_at timestamptz not null default now(),
  unique (user_id, word_key)
);

create index if not exists vocab_words_user_created on vocab_words (user_id, created_at desc);

alter table vocab_words enable row level security;

-- One review answer in one statement, so two quick answers cannot both read the old streak.
-- A learned word is left as it is (no row returned).
create or replace function vocab_word_answer(p_id uuid, p_user uuid, p_know boolean)
returns setof vocab_words
language sql
as $$
  update vocab_words
  set streak = case when p_know then least(streak + 1, 3) else 0 end,
      learned_at = case when p_know and streak + 1 >= 3 then now() else null end,
      reviewed_at = now()
  where id = p_id and user_id = p_user and learned_at is null
  returning *;
$$;
```

```ts
import { NextRequest, NextResponse } from "next/server";
import { getSessionUser } from "@/lib/session";
import { supabaseAdmin } from "@/lib/supabase";
import { isValidWord, normalizeMeaning, normalizeWord, vocabWordKey, type VocabErrorCode } from "@/lib/vocab";
import { lookupWord } from "@/lib/word-lookup";

export const dynamic = "force-dynamic";

const fail = (status: number, code: VocabErrorCode, error: string, extra: Record<string, unknown> = {}) =>
  NextResponse.json({ error, code, ...extra }, { status });

export async function GET(req: NextRequest) {
  const user = await getSessionUser(req);
  if (!user) return NextResponse.json({ error: "Unauthorized" }, { status: 401 });
  const { data, error } = await supabaseAdmin()
    .from("vocab_words")
    .select("*")
    .eq("user_id", user.id)
    .order("created_at", { ascending: false });
  if (error) return NextResponse.json({ error: error.message }, { status: 500 });
  return NextResponse.json(data ?? []);
}

export async function POST(req: NextRequest) {
  const user = await getSessionUser(req);
  if (!user) return NextResponse.json({ error: "Unauthorized" }, { status: 401 });

  const body = await req.json().catch(() => null);
  const word = typeof body?.word === "string" ? normalizeWord(body.word) : "";
  if (!isValidWord(word)) return fail(400, "invalid_word", "Invalid word");
  const meaning = normalizeMeaning(body?.meaningTr);
  if (!meaning.ok) return fail(400, "invalid_meaning", "Invalid meaning");

  const db = supabaseAdmin();
  const key = vocabWordKey(word);
  const find = () => db.from("vocab_words").select("*").eq("user_id", user.id).eq("word_key", key).maybeSingle();

  const { data: existing } = await find();
  if (existing) return fail(409, "duplicate", "Already in your list", { word: existing });

  const found = await lookupWord(word);
  const { data, error } = await db
    .from("vocab_words")
    .insert({
      user_id: user.id,
      word,
      word_key: key,
      meaning_tr: meaning.value,
      ipa: found.ipa,
      audio_url: found.audioUrl,
      senses: found.senses,
      lookup: found.lookup,
    })
    .select("*")
    .single();
  if (error?.code === "23505") {
    const { data: winner } = await find();
    return fail(409, "duplicate", "Already in your list", { word: winner });
  }
  if (error) return NextResponse.json({ error: error.message }, { status: 500 });
  return NextResponse.json(data, { status: 201 });
}
```

```ts
import { getSessionUser } from "@/lib/session";
import { supabaseAdmin } from "@/lib/supabase";
import { normalizeMeaning, type VocabErrorCode, type VocabWord } from "@/lib/vocab";
import { lookupWord } from "@/lib/word-lookup";

export const dynamic = "force-dynamic";

type Params = { params: Promise<{ id: string }> };
const CHANGES = ["meaningTr", "answer", "unlearn", "relookup"] as const;

const fail = (status: number, code: VocabErrorCode, error: string) => NextResponse.json({ error, code }, { status });

export async function PATCH(req: NextRequest, { params }: Params) {
  const user = await getSessionUser(req);
  if (!user) return NextResponse.json({ error: "Unauthorized" }, { status: 401 });

  const body = await req.json().catch(() => null);
  const sent = body && typeof body === "object" ? CHANGES.filter((k) => k in body) : [];
  if (sent.length !== 1) return fail(400, "invalid_request", "Send exactly one change");

  const { id } = await params;
  const db = supabaseAdmin();
  const { data: word } = await db.from("vocab_words").select("*").eq("id", id).eq("user_id", user.id).maybeSingle();
  if (!word) return fail(404, "not_found", "Word not found");

  const update = async (values: Record<string, unknown>) => {
    const { data, error } = await db.from("vocab_words").update(values).eq("id", id).eq("user_id", user.id).select("*").single();
    if (error) return NextResponse.json({ error: error.message }, { status: 500 });
    return NextResponse.json(data);
  };

  switch (sent[0]) {
    case "meaningTr": {
      const meaning = normalizeMeaning(body.meaningTr);
      if (!meaning.ok) return fail(400, "invalid_meaning", "Invalid meaning");
      return update({ meaning_tr: meaning.value });
    }
    case "answer": {
      if (body.answer !== "know" && body.answer !== "dontKnow") return fail(400, "invalid_request", "Unknown answer");
      const { data, error } = await db.rpc("vocab_word_answer", { p_id: id, p_user: user.id, p_know: body.answer === "know" });
      if (error) return NextResponse.json({ error: error.message }, { status: 500 });
      return NextResponse.json((data as VocabWord[] | null)?.[0] ?? word);
    }
    case "unlearn":
      return update({ streak: 0, learned_at: null });
    case "relookup": {
      const found = await lookupWord((word as VocabWord).word);
      return update({ ipa: found.ipa, audio_url: found.audioUrl, senses: found.senses, lookup: found.lookup });
    }
  }
}

export async function DELETE(req: NextRequest, { params }: Params) {
  const user = await getSessionUser(req);
  if (!user) return NextResponse.json({ error: "Unauthorized" }, { status: 401 });
  const { id } = await params;
  const { data, error } = await supabaseAdmin().from("vocab_words").delete().eq("id", id).eq("user_id", user.id).select("id");
  if (error) return NextResponse.json({ error: error.message }, { status: 500 });
  if (!data || data.length === 0) return fail(404, "not_found", "Word not found");
  return NextResponse.json({ ok: true });
}
```

```ts
// src/lib/words-client.ts
import type { VocabWord } from "@/lib/vocab";

export type WordsResult<T> =
  | { ok: true; data: T }
  | { ok: false; status: number; code: string | undefined; error: string; word?: VocabWord };

export type WordChange = { meaningTr: string } | { answer: "know" | "dontKnow" } | { unlearn: true } | { relookup: true };

async function call<T>(url: string, init?: RequestInit): Promise<WordsResult<T>> {
  let res: Response;
  try {
    res = await fetch(url, { ...init, headers: { "content-type": "application/json" } });
  } catch {
    return { ok: false, status: 0, code: undefined, error: "" };
  }
  const body = await res.json().catch(() => null);
  if (res.ok) return { ok: true, data: body as T };
  return {
    ok: false,
    status: res.status,
    code: typeof body?.code === "string" ? body.code : undefined,
    error: typeof body?.error === "string" ? body.error : "",
    ...(body?.word ? { word: body.word as VocabWord } : {}),
  };
}

export const listWords = () => call<VocabWord[]>("/api/words");
export const addWord = (word: string, meaningTr: string) =>
  call<VocabWord>("/api/words", { method: "POST", body: JSON.stringify({ word, meaningTr }) });
export const changeWord = (id: string, change: WordChange) =>
  call<VocabWord>(`/api/words/${id}`, { method: "PATCH", body: JSON.stringify(change) });
export const deleteWord = (id: string) => call<{ ok: true }>(`/api/words/${id}`, { method: "DELETE" });
```

```ts
export class ApiError extends Error {
  constructor(public status: number, message: string, public code?: string) {
    super(message);
    this.name = "ApiError";
  }
}

async function errorBody(res: Response): Promise<{ message: string; code?: string }> {
  const text = await res.text().catch(() => "");
  try {
    const parsed = JSON.parse(text);
    if (parsed && typeof parsed.error === "string") {
      return { message: parsed.error, code: typeof parsed.code === "string" ? parsed.code : undefined };
    }
  } catch {
    // non-JSON body — use the raw text below
  }
  return { message: text || `HTTP ${res.status}` };
}
```

```tsx
                  <Pressable onPress={() => answer(word, true)} className="flex-1 items-center rounded-lg bg-emerald-700 py-3">
                    <Text className="text-sm font-semibold text-white">{t.wordsKnow}</Text>
                  </Pressable>
                </View>
              </View>
            ) : (
              <View className="w-full"><PrimaryButton label={t.wordsShow} onPress={() => setState((s) => (s ? revealReview(s) : s))} /></View>
            )}
          </View>
        ) : null}
        {state && state.words.length > 0 && reviewDone(state) ? (
          <View className="items-center gap-3 rounded-xl bg-surface p-6">
            <Text className="text-center text-lg font-semibold text-gray-900">
              {fillTemplate(t.wordsRoundDone, { known: String(state.known), total: String(state.words.length), learned: String(state.learned) })}
            </Text>
            <PrimaryButton label={t.wordsAnotherRound} onPress={() => void start()} />
          </View>
        ) : null}
      </ScrollView>
    </TabScreen>
  );
}
```
